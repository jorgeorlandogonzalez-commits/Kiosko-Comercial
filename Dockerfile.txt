# ============================================================================
# Kiosko Comercial V3.5 - Dockerfile para Cloud Run
# Multi-stage build optimizado para producción
# ============================================================================

# Etapa 1: Build
FROM node:22-alpine AS builder

WORKDIR /app

# Copiar package files
COPY package*.json ./

# Instalar TODAS las dependencias (incluyendo dev)
RUN npm ci

# Copiar código fuente completo
COPY . .

# Compilar frontend y backend
RUN npm run build

# ============================================================================
# Etapa 2: Producción (imagen mínima)
# ============================================================================
FROM node:22-alpine AS production

WORKDIR /app

# Copiar package files
COPY package*.json ./

# Instalar solo dependencias de producción
RUN npm ci --omit=dev && npm cache clean --force

# Copiar build generado desde etapa builder
COPY --from=builder /app/dist ./dist

# Variables de entorno para Cloud Run
ENV NODE_ENV=production
ENV PORT=8080

# Puerto que Cloud Run usa
EXPOSE 8080

# Health check optimizado para Cloud Run
HEALTHCHECK --interval=30s --timeout=5s --start-period=30s --retries=3 \
  CMD wget --no-verbose --tries=1 --spider http://localhost:8080/api/health || exit 1

# Comando de inicio
CMD ["node", "dist/server.cjs"]
